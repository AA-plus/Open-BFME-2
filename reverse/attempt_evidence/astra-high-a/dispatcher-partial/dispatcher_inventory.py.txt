from pathlib import Path
import re,csv
s=Path('build/astra-high-a/dispatcher-wb-retail-listing.txt').read_text().split('game.dat body:')[1]
rvas={int(x,16)-0x400000 for x in re.findall(r'call 0x([0-9a-f]+)',s)}
lines=[]
with Path('reverse/functions.csv').open(newline='') as f:
 for row in csv.reader(f):
  if len(row)>3 and row[2].startswith('0x') and int(row[2],16) in rvas:lines.append(row)
Path('build/astra-high-a/dispatcher-callees.json').write_text(__import__('json').dumps(lines,indent=2))
for r in lines: print(r[2],r[0],r[4])
