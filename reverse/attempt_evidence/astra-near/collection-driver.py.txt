from astra_near_trials import ROOT,trial
import sys
base=(ROOT/'reverse/attempts/0x0059be1f.cpp').read_text(encoding='latin-1')
symbol=base.splitlines()[0][3:]
variants=[]
if sys.argv[1]=='owner':
    variants.append(('baseline',base))
    after=base.replace('      int owner=army->owner;bool found=false;\n','').replace('      enemy->push_back((const ModuleData*const&)army);','      enemy->push_back((const ModuleData*const&)army);\n      int owner=army->owner;bool found=false;')
    variants.append(('owner-after-push',after))
    variants.append(('owner-after-volatile-view',after.replace('      int owner=army->owner;', '      int owner=((const volatile Rva002B488EResult*)army)->owner;')))
    variants.append(('owner-after-same-unsigned',after.replace('      int owner=army->owner;','      unsigned int owner=army->owner;')))
    variants.append(('owner-after-inline-getter',after.replace('int owner;};','int owner; int getOwner() const {return owner;} };').replace('      int owner=army->owner;','      int owner=army->getOwner();')))
if sys.argv[1]=='scopes':
    after=(ROOT/'build/astra-near/collection/owner-after-push/c1.cpp').read_text()
    friendly='''      unsigned int owner=army->owner;bool found=false;
      for(unsigned j=0;j<players->size();++j) {
       if(*(int*)((char*)(*players)[j]+0x14)==owner) {found=true;break;}
      }
      if((found == 0)) {Rva002E2903Player*p=((Rva002BA8F1Logic*)TheLivingWorldLogic)->find(owner,0);players->push_back((const ModuleData*const&)p);}'''
    enemy=friendly.replace('unsigned int owner','int owner')
    for where in ('both','enemy'):
        for mode in ('army','owner','refarmy'):
            param={'army':'const Rva002B488EResult *army','owner':'int owner','refarmy':'const Rva002B488EResult *const &army'}[mode]
            body=enemy if mode!='owner' else enemy.replace('int owner=army->owner;','')
            helper='\nstatic __forceinline void collectOwner('+param+',PointerVector *players) {\n'+body+'\n}\n'
            call='      collectOwner('+('army->owner' if mode=='owner' else 'army')+',players);'
            text=after.replace(enemy,call)
            if where=='both':text=text.replace(friendly,call)
            text=text.replace('void LivingWorldAIArmyMover::rva0059BE1F',helper+'void LivingWorldAIArmyMover::rva0059BE1F')
            variants.append(('helper-corrected-'+where+'-'+mode,text))
    for name,head in [('postincrement','for(unsigned j=0;j<players->size();j++)'),('signed-index','for(int j=0;j<(int)players->size();++j)'),('index-long','for(unsigned long j=0;j<players->size();++j)')]:
        variants.append(('after-'+name,after.replace('for(unsigned j=0;j<players->size();++j)',head)))
    variants.append(('after-preincrement-outer',after.replace('r++)','++r)').replace('distance++)','++distance)')))
if sys.argv[1]=='distance':
    after=(ROOT/'build/astra-near/collection/owner-after-push/c1.cpp').read_text()
    start=after.index('  for(int distance=0;distance<=maxDistance;distance++) {')
    loop=after[start:after.rfind('\n }\n}')]
    params=[('IntTree&','tree'),('Rva002E2903Player*','player'),('int','distance'),('int','maxDistance'),('PointerVector*','friendly'),('PointerVector*','enemy'),('PointerVector*','players'),('Rva004FF5B7*','info'),('int','difficulty')]
    for shift in (0,1,2):
        order=params[shift:]+params[:shift]
        signature=','.join(a+' '+b for a,b in order)
        args=','.join('0' if b=='distance' else b for a,b in order)
        body=loop.replace('for(int distance=0;','for(;')
        helper='\nstatic __forceinline void collectDistanceRange('+signature+') {\n'+body+'\n}\n'
        text=after[:start]+'  collectDistanceRange('+args+');'+after[after.rfind('\n }\n}'):]
        text=text.replace('void LivingWorldAIArmyMover::rva0059BE1F',helper+'void LivingWorldAIArmyMover::rva0059BE1F')
        variants.append(('byvalue-distance-'+str(shift),text))
    t=after.replace('for(int distance=0;', 'for(volatile int distance=0;').replace('.equal_range(distance)', '.equal_range((const int &)distance)')
    variants.append(('volatile-distance',t))
if sys.argv[1]=='scopes': variants=[(n,t) for n,t in variants if n.startswith('helper-corrected')]
for name,text in variants:
    r=trial(name,text,symbol,0x59be1f,497,'collection')
    if r.get('exact'):break
