from pathlib import Path
import json, sys, re
ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / 'tools'))
import build
from variant_search import Scorer, diff_text
from explain_mismatch import disassemble_capstone

def trial(name, text, symbol, rva, size, group='retreat'):
    out = ROOT / 'build/astra-near' / group / name
    score = Scorer(symbol, rva, size, out)
    result = score(text)
    summary = {k:v for k,v in result.items() if k not in ('compiled','target')}
    if 'compiled' in result:
        raw, relocs = build.read_object_symbol_bytes(out/'c1.obj', symbol, size)
        summary['raw_size'] = len(raw)
        summary['different_offsets'] = [i for i,(a,b) in enumerate(zip(result['compiled'],result['target'])) if a!=b]
        summary['unresolved'] = build.compile_function({'name':symbol,'target_rva':hex(rva),'target_size':str(size),'source':(out/'c1.cpp').relative_to(ROOT).as_posix(),'notes':''},score.symbol_map,out/'c1.obj')['unresolved']
        for k in ('compiled','target'):
            (out/(k+'.bin')).write_bytes(result[k])
            (out/(k+'.txt')).write_text('\n'.join(disassemble_capstone(result[k],rva)),encoding='utf-8')
        (out/'raw.txt').write_text('\n'.join(disassemble_capstone(raw,rva)),encoding='utf-8')
        (out/'relocs.json').write_text(json.dumps(relocs,indent=2))
        (out/'diff.txt').write_text(diff_text(result['compiled'],result['target'],300))
    (out/'result.json').write_text(json.dumps(summary,indent=2))
    console=dict(summary)
    if len(console.get('different_offsets',[]))>15: console['different_offsets']=str(len(console['different_offsets']))+' differences'
    print(name, json.dumps(console), flush=True)
    return result
