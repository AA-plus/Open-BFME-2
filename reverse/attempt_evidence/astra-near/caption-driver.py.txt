from astra_near_trials import ROOT, trial
import sys
base=(ROOT/'reverse/attempts/0x00278064.cpp').read_text(encoding='latin-1')
symbol='?setCaptionText@Drawable@@QAEXABVUnicodeString@@@Z'
variants=[]
if sys.argv[1]=='storage':
    variants.append(('baseline',base))
    for typ in ('volatile bool','bool'):
        variants.append(('aligned4-'+typ.replace(' ','-'),base.replace('volatile bool bold','__declspec(align(4)) '+typ+' bold')))
    for typ in ('const AsciiString &','AsciiString '):
        for align in ('','__declspec(align(4)) '):
            start=base.index('        volatile bool bold')
            end=base.index('        m_caption->setFont',start)
            text=base[:start]+'''        GameFont *font;
        {
            '''+typ+'''name = ((Rva00274DB6AsciiField *)TheInGameUI)->get();
            '''+align+'''bool bold = ((CaptionUIFields *)TheInGameUI)->bold;
            font = TheFontLibrary->getFont(&name,
                (float)TheGlobalLanguageData->adjustFontSize(((CaptionUIFields *)TheInGameUI)->pointSize), bold);
        }
'''+base[end:]
            variants.append(('name-'+('ref' if '&' in typ else 'value')+('-aligned4' if align else ''),text))
if sys.argv[1]=='getters':
    clean=base.replace('        volatile bool bold = ((CaptionUIFields *)TheInGameUI)->bold;\n','').replace('pointSize), bold);','pointSize), ((CaptionUIFields *)TheInGameUI)->bold);')
    for ret in ('bool','const bool','const bool &'):
        t=clean.replace('    bool bold;','    bool bold;\n    '+ret+' isBold() const { return bold; }').replace('((CaptionUIFields *)TheInGameUI)->bold);','((CaptionUIFields *)TheInGameUI)->isBold());')
        variants.append(('bold-getter-'+ret.replace(' ','-').replace('&','ref'),t))
    variants.append(('sanitized-align4',clean.replace('    UnicodeString sanitized','    __declspec(align(4)) UnicodeString sanitized')))
if sys.argv[1]=='both':
    clean=(ROOT/'build/astra-near/caption/bold-getter-bool/c1.cpp').read_text(encoding='latin-1')
    clean=clean.replace('    int pointSize;','    int pointSize;\n    int getPointSize() const { return pointSize; }').replace('->pointSize)', '->getPointSize())')
    variants.append(('both-getters',clean))
    variants.append(('both-getters-nonconst',clean.replace('isBold() const','isBold()').replace('getPointSize() const','getPointSize()')))
    for align in ('','__declspec(align(4)) '):
        s=clean.replace('    UnicodeString sanitized = captionText;', '    '+align+'struct SanitizedStorage { UnicodeString value; SanitizedStorage(const UnicodeString &s):value(s){} } storage(captionText);\n    UnicodeString &sanitized = storage.value;')
        variants.append(('both-storage-struct'+('-align4' if align else ''),s))
    for flag in ('/G6','/G5','/Ob1','/Ob2','/Og','/Oi','/Os'):
        variants.append(('both-flag-'+flag[1:],clean.replace('/G7',flag) if flag.startswith('/G') else clean.replace('// cl: ','// cl: '+flag+' ')))
if sys.argv[1]=='lifetime':
    clean=(ROOT/'build/astra-near/caption/both-getters/c1.cpp').read_text(encoding='latin-1')
    for typ in ('UnicodeString','UnicodeString &'):
        start=clean.index('void Drawable::setCaptionText(const UnicodeString &captionText)\n{')
        head=clean[:start].replace('    void setCaptionText(const UnicodeString &captionText);','    void setCaptionText(const UnicodeString &captionText);\n    __forceinline void setSanitizedCaption('+typ+' sanitized);')
        body=clean[start:]
        split=body.index('    TheLanguageFilter->filterLine(sanitized);')
        helper='__forceinline void Drawable::setSanitizedCaption('+typ+' sanitized)\n{\n'+body[split:]
        outer=body[:split]
        if typ=='UnicodeString': outer=outer.replace('    UnicodeString sanitized = captionText;','    setSanitizedCaption(captionText);')
        else: outer+='    setSanitizedCaption(sanitized);\n'
        variants.append(('inline-sanitized-'+('ref' if '&' in typ else 'value'),head+helper+'\n'+outer+'}\n'))
    for mode,replacement in [('init-parens','UnicodeString sanitized(captionText);'),('source-pointer','const UnicodeString *source = &captionText;\n    UnicodeString sanitized(*source);'),('source-reference','const UnicodeString &source = captionText;\n    UnicodeString sanitized(source);'),('one-array','UnicodeString stored[1] = {captionText};\n    UnicodeString &sanitized = stored[0];')]:
        variants.append((mode,clean.replace('UnicodeString sanitized = captionText;',replacement)))
    # A fresh block for the input guard leaves its reference local dead before copying.
    text=clean.replace('    if (captionText.isEmpty())','    {\n    const UnicodeString &input = captionText;\n    if (input.isEmpty())').replace('    UnicodeString sanitized = captionText;','    }\n    UnicodeString sanitized = captionText;')
    variants.append(('guard-scope',text))
    t=(ROOT/'build/astra-near/caption/both-storage-struct/c1.cpp').read_text(encoding='latin-1').replace('SanitizedStorage(const UnicodeString','__forceinline SanitizedStorage(const UnicodeString')
    variants.append(('storage-forceinline',t))
for name,text in variants:
    r=trial(name,text,symbol,0x278064,300,'caption')
    if r.get('exact'): break
